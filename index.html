// ========================================================================== // Invoice → Zoho Books uploader — single-file worker for Cloudflare dashboard // v2: auto-creates missing contacts, strips undefined values, better errors // ========================================================================== const ALLOWED_ORIGIN = '*'; // tighten to your GitHub Pages URL when ready const cors = { 'Access-Control-Allow-Origin': ALLOWED_ORIGIN, 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, X-App-Secret', }; const json = (data, status = 200) => new Response(JSON.stringify(data), { status, headers: { 'Content-Type': 'application/json', ...cors }, }); // ---------- Gemini extraction ---------- // Allowed models — prevents arbitrary model strings from being sent to Google. const ALLOWED_GEMINI_MODELS = [ 'gemini-2.5-flash', 'gemini-2.5-flash-lite', 'gemini-2.0-flash-lite', 'gemini-flash-latest', ]; const DEFAULT_GEMINI_MODEL = 'gemini-2.5-flash'; const geminiEndpoint = (model) => `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`; const SCHEMA = { type: 'OBJECT', properties: { party_name: { type: 'STRING' }, party_gstin: { type: 'STRING' }, document_number: { type: 'STRING' }, document_date: { type: 'STRING' }, due_date: { type: 'STRING' }, currency: { type: 'STRING' }, line_items: { type: 'ARRAY', items: { type: 'OBJECT', properties: { description: { type: 'STRING' }, hsn_sac: { type: 'STRING' }, quantity: { type: 'NUMBER' }, unit: { type: 'STRING' }, rate: { type: 'NUMBER', description: 'Per-unit pre-tax price. Identify the CUSTOMER as party_name.'; const prompt = `Extract structured data from this ${kind === 'bill' ? 'vendor bill' : 'sales invoice'}. ${partyHint} CRITICAL extraction rules: - Dates must be in YYYY-MM-DD format. - Numbers must be plain numbers (no currency symbols, no thousands separators). - For each line_item, "amount" MUST be the PRE-TAX line total (taxable_value + non_taxable_value, NOT the total-including-tax column). - If a line has only a Non-Taxable / Exempt value with no Taxable value (e.g. airport charges, exempt items), set tax_percent to 0 and amount to the non-taxable value. - If the invoice does not show a per-unit rate (lump sum charges), omit "rate" entirely and set quantity to 1. - For Indian invoices: capture GSTIN, HSN/SAC codes. tax_percent is the SUM of CGST+SGST (for intra-state) or IGST (for inter-state). E.g. CGST 2.5% + SGST 2.5% → tax_percent: 5. - If a field is not present, omit it (do not invent values).`; const body = { contents: [{ parts: [ { text: prompt }, { inline_data: { mime_type: mimeType, data: fileBase64 } }, ], }], generationConfig: { response_mime_type: 'application/json', response_schema: SCHEMA, temperature: 0, }, }; throw new Error(`Gemini error ${res.status}: ${text}`); } const data = await res.json(); const text = data?.candidates?.[0]?.content?.parts?.[0]?.text; if (!text) throw new Error('Gemini returned no content'); return JSON.parse(text); } // ---------- Zoho Books (India DC) ---------- const ZOHO_ACCOUNTS = 'https://accounts.zoho.in'; const ZOHO_API = 'https://www.zohoapis.in/books/v3'; async function getAccessToken(env) { const params = new URLSearchParams({ refresh_token: env.ZOHO_REFRESH_TOKEN, client_id: env.ZOHO_CLIENT_ID, client_secret: env.ZOHO_CLIENT_SECRET, grant_type: 'refresh_token', }); throw new Error(`Zoho API ${res.status}: ${data.message || JSON.stringify(data)}${sentBody}`); } return data; } async function searchContact(env, name, type) { const path = `/contacts?contact_name_contains=${encodeURIComponent(name)}&contact_type=${type}`; const data = await zohoFetch(env, path); return data.contacts || []; } async function createContact(env, contactPayload) { // contactPayload is a Zoho Books contact object. // Required: contact_name, contact_type ('vendor' or 'customer'). // For Indian orgs with GST: gst_treatment is required. // 'business_gst' → gst_no is also required (15 char GSTIN) // 'business_none' / 'consumer' / 'overseas' → no gst_no needed // Optional: contact_persons array with first_name, last_name, email. const data = await zohoFetch(env, '/contacts', { method: 'POST', body: JSON.stringify(contactPayload), }); return data.contact; } // Build a contact payload from form fields, with India GST validation. function buildContactPayload({ name, type, gst_treatment, gst_no, first_name, last_name, email }) { if (!name) throw new Error('Vendor/customer name is required'); if (!type) throw new Error('Contact type is required'); // Normalize GSTIN before any logic uses it if (gst_no) gst_no = gst_no.toUpperCase().trim(); // Auto-correct: GSTINs starting with "99" are OIDAR / non-resident foreign // service providers (e.g. Apple Ireland, Google Cloud, Microsoft Ireland). // They have no Indian state code, so Zoho rejects business_gst treatment // with "Please provide a valid state code". Force overseas treatment. if (gst_no && gst_no.startsWith('99') && gst_treatment === 'business_gst') { gst_treatment = 'overseas'; } if (!gst_treatment) throw new Error('GST Treatment is required'); const validTreatments = ['business_gst', 'business_none', 'overseas', 'consumer', 'deemed_export', 'sez', 'sez_developer']; if (!validTreatments.includes(gst_treatment)) { throw new Error(`Invalid GST Treatment. Must be one of: ${validTreatments.join(', ')}`); } if (gst_treatment === 'business_gst' && !gst_no) { throw new Error('GSTIN is required when GST Treatment is "Registered Business"'); } // 15-char alphanumeric format check applies to any provided GSTIN if (gst_no && !/^[0-9A-Z]{15}$/.test(gst_no)) { throw new Error('GSTIN must be exactly 15 characters (uppercase letters and digits)'); } // For business_gst, first 2 chars must be a valid Indian state code (01-38). // Catches the case where someone manually selects "Registered Business" but // pastes a 99-prefix OIDAR GSTIN — fails locally with a useful hint before // Zoho returns the generic "valid state code" error. if (gst_treatment === 'business_gst') { const stateCode = parseInt(gst_no.substring(0, 2), 10); if (isNaN(stateCode) || stateCode < 1 || stateCode > 38) { throw new Error( `GSTIN "${gst_no}" starts with "${gst_no.substring(0, 2)}", which is not a valid Indian state code. ` + `If this is a foreign / OIDAR vendor (Apple Ireland, Google, Microsoft, etc.), select "Overseas" as GST Treatment.` ); } } const payload = { contact_name: name, contact_type: type, gst_treatment, }; // Send GSTIN for both business_gst and overseas. Omit for other treatments. if (gst_no && (gst_treatment === 'business_gst' || gst_treatment === 'overseas')) { payload.gst_no = gst_no; } if (first_name || last_name || email) { payload.contact_persons = [{ first_name: first_name || '', last_name: last_name || '', email: email || '', is_primary_contact: true, }]; } return payload; } async function ensureContact(env, payload, kind) { const idField = kind === 'bill' ? 'vendor_id' : 'customer_id'; const nameField = kind === 'bill' ? 'vendor_name' : 'customer_name'; const contactType = kind === 'bill' ? 'vendor' : 'customer'; if (payload[idField]) return payload; const name = payload[nameField]; if (!name) throw new Error(`Missing ${nameField} and ${idField}`); const matches = await searchContact(env, name, contactType); if (matches.length > 0) { payload[idField] = matches[0].contact_id; } else { // Auto-create a minimal contact when pushing a bill/invoice without a prior selection. // Defaults to 'business_none' (unregistered) to satisfy India GST requirements. // For full vendor setup with GSTIN, use the "Add new vendor" form instead. const created = await createContact(env, { contact_name: name, contact_type: contactType, gst_treatment: 'business_none', }); payload[idField] = created.contact_id; } delete payload[nameField]; return payload; } // Strip undefined / null / empty-string values recursively function clean(obj) { if (Array.isArray(obj)) return obj.map(clean); if (obj && typeof obj === 'object') { const out = {}; for (const [k, v] of Object.entries(obj)) { if (v === undefined || v === null || v === '') continue; out[k] = clean(v); } return out; } return obj; } function applyDefaultAccount(cleaned, accountId) { if (!accountId) return; cleaned.line_items = (cleaned.line_items || []).map((li) => ({ ...li, account_id: li.account_id || accountId, })); } // Cache for tax exemption code → id mapping. Refreshed once per worker invocation. let _exemptionMap = null; async function getExemptionMap(env) { if (_exemptionMap) return _exemptionMap; const list = await listTaxExemptions(env); _exemptionMap = {}; for (const ex of list) { const code = ex.tax_exemption_code || ex.exemption_reason || ex.name; if (code) _exemptionMap[code] = ex.tax_exemption_id; } return _exemptionMap; } // Resolve every line item to either a tax_id (for taxable lines) or a // tax_exemption_id (for zero-tax lines). // Selects GST vs IGST based on whether the vendor is in the same state as our org. async function applyTaxResolution(env, cleaned) { const items = cleaned.line_items || []; const needsExemption = items.some((li) => { const tp = parseFloat(li.tax_percentage); return !tp || tp === 0; }); const exemptMap = needsExemption ? await getExemptionMap(env) : {}; // Determine inter/intra-state from party GSTIN (passed in cleaned._party_gstin) vs org home state. const orgState = env.ZOHO_HOME_STATE_CODE; // e.g. "23" for MP const partyState = gstinStateCode(cleaned._party_gstin); // If we can't determine the party's state (no GSTIN extracted), default to intra-state // since that's the safer assumption for a domestic small-business workflow. const isInterState = orgState && partyState && partyState !== orgState; const resolvedItems = []; for (let i = 0; i < items.length; i++) { const li = items[i]; const tp = parseFloat(li.tax_percentage); if (tp && tp > 0) { const taxId = await resolveTaxId(env, tp, isInterState); if (!taxId) { throw new Error(`Line ${i + 1} ("${li.name || 'unnamed'}"): no tax record found for ${tp}%. Create one in Zoho Settings → Taxes.`); } const { tax_percentage, tax_exemption_code, tax_exemption_id, ...rest } = li; resolvedItems.push({ ...rest, tax_id: taxId }); } else { if (!li.tax_exemption_code && !li.tax_exemption_id) { throw new Error(`Line ${i + 1} ("${li.name || 'unnamed'}") has 0% tax but no exemption reason.`); } let exemptionId = li.tax_exemption_id; if (!exemptionId && li.tax_exemption_code) { exemptionId = exemptMap[li.tax_exemption_code]; if (!exemptionId) { throw new Error(`Line ${i + 1}: exemption "${li.tax_exemption_code}" not found. Available: ${Object.keys(exemptMap).join(', ')}`); } } const { tax_percentage, tax_exemption_code, ...rest } = li; resolvedItems.push({ ...rest, tax_exemption_id: exemptionId }); } } cleaned.line_items = resolvedItems; // Strip our internal-only field before sending to Zoho delete cleaned._party_gstin; } async function listTaxExemptions(env) { const data = await zohoFetch(env, '/settings/taxexemptions'); return data.tax_exemptions || []; } async function listTaxes(env) { const data = await zohoFetch(env, '/settings/taxes'); return data.taxes || []; } // Cache for taxes list. let _taxList = null; async function getTaxList(env) { if (_taxList) return _taxList; _taxList = await listTaxes(env); return _taxList; } // Pick the right tax for a given percentage and transaction type. // isInterState=true → prefer IGST*. isInterState=false → prefer GST*/CGST*/SGST*. async function resolveTaxId(env, percentage, isInterState) { const taxes = await getTaxList(env); const candidates = taxes.filter((t) => parseFloat(t.tax_percentage) === percentage); if (candidates.length === 0) return null; // Heuristic: name starting with "IGST" is inter-state; anything else is intra-state. const inter = candidates.find((t) => /^igst/i.test(t.tax_name)); const intra = candidates.find((t) => !/^igst/i.test(t.tax_name)); if (isInterState) return (inter || intra)?.tax_id || null; return (intra || inter)?.tax_id || null; } // Indian GSTIN first 2 digits = state code (e.g. "23" = MP, "27" = MH, "29" = KA). function gstinStateCode(gstin) { if (!gstin || typeof gstin !== 'string') return null; const m = gstin.trim().match(/^(\d{2})/); return m ? m[1] : null; } async function attachToBill(env, billId, fileBase64, mimeType, filename) { // Zoho's attachment endpoint requires multipart/form-data, not JSON. // We construct it manually because Cloudflare Workers' FormData with Blob // sometimes loses the filename. The body is a multipart/form-data buffer. const token = await getAccessToken(env); const url = new URL(`${ZOHO_API}/bills/${billId}/attachment`); url.searchParams.set('organization_id', env.ZOHO_ORG_ID); } async function createBill(env, payload) { const cleaned = clean(payload); applyDefaultAccount(cleaned, env.ZOHO_DEFAULT_EXPENSE_ACCOUNT_ID); await applyTaxResolution(env, cleaned); await ensureContact(env, cleaned, 'bill');
